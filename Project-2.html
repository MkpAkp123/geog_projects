mean_center <- function(x, y, w = rep(1, length(x))) {
  tibble(x = sum(x * w) / sum(w),
         y = sum(y*w)/sum(w))
}

standard_distance <- function(x, y, w = rep(1, length(x))) {
  center <- mean_center(x, y, w)
  sqrt(sum(w * ((x - center$x)^2 + (y - center$y)^2)) / sum(w))
}
combined_kids_xy <- combined_kids |>
  st_centroid()|>
  st_coordinates()
## Warning: st_centroid assumes attributes are constant over geometries
combined_kids <- combined_kids |>
  mutate(x = combined_kids_xy[, "X"],
         y = combined_kids_xy[, "Y"])

sd <- standard_distance(combined_kids$x, combined_kids$y)

center <- mean_center(combined_kids$x,combined_kids$y,w=combined_kids$estimate.x)

poverty_center <- st_as_sf(center,coords=c("x","y"),crs=st_crs(combined_kids))
poverty_circle <- st_buffer(poverty_center,sd)
poverty_circle <- poverty_circle |>
  mutate(Type="Center of Child Poverty")
